\begin{abstract}
This report details the statistical formulation for fitting a parametric spiral vector field to noisy angular data. We derive the vector field from a scalar potential function, analyze the heteroskedastic noise structure induced by the Cartesian-to-Polar transformation, and formulate a Weighted Least Squares (WLS) objective function to estimate the model parameters.
\end{abstract}

\section{Introduction}
The objective of this model is to recover the parameters of a spiral vector field given a set of spatial observations. Unlike standard regression problems where errors are often assumed to be homoskedastic and Euclidean, this problem involves directional data where the noise variance depends on the magnitude of the underlying field.

\section{Data and Preprocessing}
The dataset consists of spatial coordinates $Z_i = (x_i, y_i)$ and observed vectors $W_i \in \mathbb{R}^2$. Preprocessing involves converting Cartesian vector observations into angular measurements $\hat{\alpha}_i$ and calculating the radial distances $r_i = \|Z_i\|$ for each observation point.

\section{Model Specification}

\subsection{Parametric Spiral Vector Field}
We define the vector field $F_\theta$ as the gradient of a scalar potential function $V_\theta(r, \phi)$, defined in polar coordinates $(r, \phi)$. The potential is given by:
\begin{equation}
V_\theta(r,\phi) = A\,g_p(r) - b\,\phi,
\end{equation}
where $A, b \in \mathbb{R}$ are coefficients, and the radial function $g_p(r)$ is defined as:
\begin{equation}
g_p(r) =
\begin{cases}
\ln r, & p = 0, \\[6pt]
\dfrac{r^p}{p}, & p \neq 0.
\end{cases}
\end{equation}
The parameter set is $\theta = \{A, b, p\}$. The induced vector field is the gradient of the potential. In polar coordinates, the gradient operator is $\nabla = \mathbf{e}_r \frac{\partial}{\partial r} + \mathbf{e}_\phi \frac{1}{r} \frac{\partial}{\partial \phi}$. Thus:
\begin{align}
F_\theta(r,\phi) &= \nabla V_\theta(r,\phi) \nonumber \\
&= \left( \frac{\partial}{\partial r} \left( A g_p(r) - b\phi \right) \right) \mathbf{e}_r + \left( \frac{1}{r} \frac{\partial}{\partial \phi} \left( A g_p(r) - b\phi \right) \right) \mathbf{e}_\phi \nonumber \\
&= A g_p'(r) \mathbf{e}_r - \frac{b}{r} \mathbf{e}_\phi.
\end{align}
Note that for all $p$, $g_p'(r) = r^{p-1}$. The magnitude of the field, denoted $\rho(r)$, is:
\begin{equation}
\rho(r) = \| F_\theta(r,\phi) \| = \sqrt{ A^2 \big(r^{p-1}\big)^2 + \frac{b^2}{r^2} }.
\end{equation}

\subsection{Observation Model and Noise Structure}
We observe the vector field at discrete spatial points $Z_i$ for $i=1, \dots, n$. The observed vector $W_i$ is the true field perturbed by isotropic Gaussian noise:
\begin{align}
W_i &= F_{\theta_0}(Z_i) + \varepsilon_i, \\
\varepsilon_i &\sim \mathcal{N}(0, \sigma^2 I_2),
\end{align}
where $I_2$ is the $2 \times 2$ identity matrix.

Our primary quantity of interest is the direction of the field. The observed angle $\hat{\alpha}_i$ is:
\begin{equation}
\hat{\alpha}_i = \operatorname{atan2}(W_{i,2},\; W_{i,1}).
\end{equation}
By linearizing the transformation from Cartesian vectors to polar angles, we find that the angular noise is heteroskedastic. Specifically, the relationship between the observed angle and the true angle $\alpha_{\theta_0}(Z_i)$ is:
\begin{equation}
\hat{\alpha}_i \approx \alpha_{\theta_0}(Z_i) + \eta_i,
\end{equation}

where the variance of the angular error $\eta_i$ is inversely proportional to the squared magnitude of the field at that point:
\begin{equation}
\operatorname{Var}(\eta_i) \approx \frac{\sigma^2}{\rho(Z_i)^2}.
\end{equation}

\begin{theorem}[Density of the angle $\Theta=\operatorname{atan2}(W_2,W_1)$ for an isotropic Gaussian]
\label{thm:projected-normal-angle}
Let $W=(W_1,W_2)^\top \sim \mathcal N_2(\mu,\sigma^2 I_2)$ with $\sigma>0$ and $\mu\in\mathbb R^2$.
Define
\[
\Theta := \operatorname{atan2}(W_2,W_1)\in(-\pi,\pi].
\]
Write $m:=\|\mu\|$ and $\phi_0:=\operatorname{atan2}(\mu_2,\mu_1)$, so that
$\mu = m(\cos\phi_0,\sin\phi_0)^\top$.
Then $\Theta$ has a density on $(-\pi,\pi]$ given by
\begin{align} \label{Angular Gaussian pdf}
f_\Theta(\theta)
&=
\frac{1}{2\pi}\exp\!\left(-\frac{m^2}{2\sigma^2}\right)
+
\frac{m\cos(\theta-\phi_0)}{\sqrt{2\pi}\,\sigma}\,
\exp\!\left(-\frac{m^2\sin^2(\theta-\phi_0)}{2\sigma^2}\right)\,
\Phi\!\left(\frac{m\cos(\theta-\phi_0)}{\sigma}\right),
\\ & \theta\in(-\pi,\pi],  
\end{align}

where $\Phi$ denotes the standard normal cumulative distribution function.
\end{theorem}

\begin{proof}
Let $w=(w_1,w_2)^\top\in\mathbb R^2$. The density of $W$ is
\[
f_W(w)=\frac{1}{2\pi\sigma^2}\exp\!\left(-\frac{\|w-\mu\|^2}{2\sigma^2}\right).
\]
Introduce polar coordinates $w=(r\cos\theta,r\sin\theta)^\top$ with $r\ge 0$ and
$\theta\in(-\pi,\pi]$, so that $\Theta=\theta$ and the Jacobian determinant is $r$.
Hence the joint density of $(R,\Theta)$ is
\[
f_{R,\Theta}(r,\theta)
=
f_W(r\cos\theta,r\sin\theta)\,r
=
\frac{1}{2\pi\sigma^2}\exp\!\left(-\frac{\|(r\cos\theta,r\sin\theta)-\mu\|^2}{2\sigma^2}\right)r.
\]
Write $\mu=m(\cos\phi_0,\sin\phi_0)^\top$. Then
\begin{align*}
\|(r\cos\theta,r\sin\theta)-\mu\|^2
&= r^2+m^2-2rm(\cos\theta\cos\phi_0+\sin\theta\sin\phi_0)\\
&= r^2+m^2-2rm\cos(\theta-\phi_0).
\end{align*}
Set $c(\theta):=\cos(\theta-\phi_0)$ and rewrite
\[
r^2+m^2-2rmc(\theta)
=
(r-mc(\theta))^2+m^2(1-c(\theta)^2)
=
(r-mc(\theta))^2+m^2\sin^2(\theta-\phi_0).
\]
Therefore,
\[
f_\Theta(\theta)
=
\int_0^\infty f_{R,\Theta}(r,\theta)\,dr
=
\frac{1}{2\pi\sigma^2}\exp\!\left(-\frac{m^2\sin^2(\theta-\phi_0)}{2\sigma^2}\right)
\int_0^\infty r\exp\!\left(-\frac{(r-mc(\theta))^2}{2\sigma^2}\right)\,dr.
\]
Let $a:=mc(\theta)$ and substitute $u=(r-a)/\sigma$ (so $r=\sigma u+a$, $dr=\sigma du$), giving
\begin{align*}
\int_0^\infty r\exp\!\left(-\frac{(r-a)^2}{2\sigma^2}\right)\,dr
&=
\int_{-a/\sigma}^{\infty} (\sigma u+a)e^{-u^2/2}\,\sigma du \\
&=
\sigma^2\int_{-a/\sigma}^{\infty} u e^{-u^2/2}\,du
\;+\;
a\sigma\int_{-a/\sigma}^{\infty} e^{-u^2/2}\,du.
\end{align*}
The first integral evaluates to $e^{-a^2/(2\sigma^2)}$ since
$\int u e^{-u^2/2}du=-e^{-u^2/2}$, hence
\[
\int_{-a/\sigma}^{\infty} u e^{-u^2/2}\,du
=
\left[-e^{-u^2/2}\right]_{-a/\sigma}^{\infty}
=
e^{-a^2/(2\sigma^2)}.
\]
For the second integral,
\[
\int_{-a/\sigma}^{\infty} e^{-u^2/2}\,du
=
\sqrt{2\pi}\,\Phi\!\left(\frac{a}{\sigma}\right),
\]
by the definition of $\Phi$ and symmetry of the standard normal density. Combining,
\[
\int_0^\infty r\exp\!\left(-\frac{(r-a)^2}{2\sigma^2}\right)\,dr
=
\sigma^2 e^{-a^2/(2\sigma^2)}
+
a\sigma\sqrt{2\pi}\,\Phi\!\left(\frac{a}{\sigma}\right).
\]
Substituting back into $f_\Theta(\theta)$ with $a=mc(\theta)$ yields
\begin{align*}
f_\Theta(\theta)
&=
\frac{1}{2\pi\sigma^2}\exp\!\left(-\frac{m^2\sin^2(\theta-\phi_0)}{2\sigma^2}\right)
\left[
\sigma^2 e^{-\frac{m^2c(\theta)^2}{2\sigma^2}}
+
(mc(\theta))\sigma\sqrt{2\pi}\,\Phi\!\left(\frac{mc(\theta)}{\sigma}\right)
\right]\\
&=
\frac{1}{2\pi}\exp\!\left(-\frac{m^2}{2\sigma^2}\right)
+
\frac{mc(\theta)}{\sqrt{2\pi}\,\sigma}
\exp\!\left(-\frac{m^2\sin^2(\theta-\phi_0)}{2\sigma^2}\right)
\Phi\!\left(\frac{mc(\theta)}{\sigma}\right),
\end{align*}
and recalling $c(\theta)=\cos(\theta-\phi_0)$ gives the stated formula.
\end{proof}

Maximizing the likelihood from Theorem~\ref{thm:projected-normal-angle} is complex due to the non-linear form of the density. However, for practical purposes, we focus on the variance structure derived from the linear approximation. Also, changing the distance structure (e.g., using sine-wrapped loss) leads to similar conclusions regarding the weighting of observations.


This implies that angular observations are less reliable in regions where the vector field magnitude $\rho(r)$ is small.

\section{Estimation and Inference}

\subsection{Weighted Least Squares Objective}
To estimate $\theta$, we minimize the discrepancy between the observed angles and the model-predicted angles. A standard least squares approach is insufficient due to the heteroskedasticity described above. We instead employ a Weighted Least Squares (WLS) approach.

The negative log-likelihood (ignoring constant terms) suggests minimizing the variance-weighted squared error. Since angular data lies on a circle (manifold $S^1$), we use a wrapped loss function based on the sine of the angular difference.

The objective function $L(\theta)$ is:
\begin{equation} \label{radial-objective}
L(\theta) = \sum_{i=1}^n \rho(Z_i)^2 \sin^2 \big( \hat{\alpha}_i - \alpha_\theta(Z_i) \big).
\end{equation}
Here, the weight $\rho(Z_i)^2$ naturally down-weights observations where the field magnitude is small (and thus angular uncertainty is high).

\subsection{Optimization}
The estimator $\hat{\theta}$ is obtained by:
\begin{equation}
\hat{\theta} = \arg\min_\theta L(\theta).
\end{equation}
Since $L(\theta)$ is non-linear with respect to parameters $p$ and the geometry of the angle $\alpha_\theta$, we utilize numerical optimization (e.g., L-BFGS-B or Newton-Raphson) to solve for $\hat{\theta}$.

\subsection{Objective correction}
As seen in Equation \eqref{radial-objective}, the weights $\rho(Z_i)^2$ provide a natural correction for the heteroskedasticity in angular observations. But this is not enough. To see that, we have to look at the log-likelihood function more closely. 
Depending on the distance function used to measure the angular error, we can derive different forms of the log-likelihood function. First, let's consider the squared error distance function.

\paragraph{Squared-error angular likelihood}
The log-likelihood under the linearized Gaussian angle model is
\begin{equation}
\ell_{\text{SE}}(\theta| \alpha_1, \ldots, \alpha_n)
= -\frac{1}{2}\sum_{i=1}^n
\left[
\log\!\left( 2\pi\,\frac{\sigma^2}{\rho_\theta(Z_i)^2} \right)
+ \frac{\big(\Delta\alpha_i(\theta)\big)^2}{\sigma^2/\rho_\theta(Z_i)^2}
\right],
\end{equation}
where $\Delta\alpha_i(\theta) = \operatorname{wrap}\big(\hat{\alpha}_i - \alpha_\theta(Z_i)\big)$ and $\rho_\theta(Z_i)$ is the field magnitude at $Z_i$.
Dropping the constant $\log(2\pi\sigma^2)$ and multiplying through gives the equivalent minimization:
\begin{equation}
\hat{\theta}
= \arg\min_\theta\;
\sum_{i=1}^n
\left[
\rho_\theta(Z_i)^2\,\big(\Delta\alpha_i(\theta)\big)^2
- \log \rho_\theta(Z_i)^2
\right].
\end{equation}
Compared to Equation~\eqref{radial-objective}, this adds a correction term $-\sum_i \log \rho_\theta(Z_i)^2$ arising from the normalization of the heteroskedastic Gaussian.

\paragraph{Sine-wrapped loss and small-angle approximation}
Using the circle-aware loss $\sin^2\Delta\alpha_i(\theta)$, the same normalization leads to
\begin{equation}
\hat{\theta}
= \arg\min_\theta\;
\sum_{i=1}^n
\left[
\rho_\theta(Z_i)^2\,\sin^2\!\big(\Delta\alpha_i(\theta)\big)
- \log \rho_\theta(Z_i)^2
\right],
\end{equation}
which coincides with the squared-error form for small angles since $\sin^2 x \approx x^2$.

\paragraph{Von Mises alternative}
A circular likelihood with concentration $\kappa_i \propto \rho_\theta(Z_i)^2/\sigma^2$ gives
\begin{equation}
\ell_{\text{VM}}(\theta)
= \sum_{i=1}^n \left[ \kappa_i(\theta)\,\cos\big(\Delta\alpha_i(\theta)\big) - \log I_0\big(\kappa_i(\theta)\big) \right],
\quad
\kappa_i(\theta) = c\,\frac{\rho_\theta(Z_i)^2}{\sigma^2},
\end{equation}
where $I_0$ is the modified Bessel function and $c>0$ is a scale constant.
Maximizing $\ell_{\text{VM}}$ yields a similar trade-off: a data-fit term and a normalization term $\log I_0(\kappa_i)$ that depends on $\rho_\theta(Z_i)^2$.
